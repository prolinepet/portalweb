FUNCTION urlEncode RETURNS CHARACTER (pcValue AS CHARACTER):
  DEFINE VARIABLE cIn  AS CHARACTER NO-UNDO.
  DEFINE VARIABLE cOut AS CHARACTER NO-UNDO.
  DEFINE VARIABLE i    AS INTEGER   NO-UNDO.
  DEFINE VARIABLE ch   AS CHARACTER NO-UNDO.
  DEFINE VARIABLE iAsc AS INTEGER   NO-UNDO.

  cIn = (IF pcValue = ? THEN "" ELSE pcValue).
  cOut = "".

  DO i = 1 TO LENGTH(cIn):
    ch = SUBSTRING(cIn, i, 1).
    iAsc = ASC(ch).

    IF (iAsc >= 48 AND iAsc <= 57) /* 0-9 */
       OR (iAsc >= 65 AND iAsc <= 90) /* A-Z */
       OR (iAsc >= 97 AND iAsc <= 122) /* a-z */ THEN
      cOut = cOut + ch.
    ELSE IF ch = " " THEN
      cOut = cOut + "%20".
    ELSE DO:
      cOut = cOut + "%" + STRING(iAsc, "HH").
    END.
  END.

  RETURN cOut.
END FUNCTION.

    DEFINE OUTPUT PARAMETER poJsonArray   AS JsonArray  NO-UNDO.

    DEFINE VARIABLE oParsed AS Progress.Lang.Object NO-UNDO.

    ASSIGN poJsonObject = ?
           poJsonArray  = ?.

    IF plcResponse = "" THEN
        RETURN.

    goParser = NEW ObjectModelParser().
    oParsed = goParser:Parse(plcResponse) NO-ERROR.
    IF ERROR-STATUS:ERROR OR NOT VALID-OBJECT(oParsed) THEN
        RETURN.

    poJsonObject = CAST(oParsed, JsonObject) NO-ERROR.
    IF VALID-OBJECT(poJsonObject) THEN
        RETURN.

    poJsonArray = CAST(oParsed, JsonArray) NO-ERROR.
END PROCEDURE.

PROCEDURE pi-portal-call:
  DEFINE INPUT  PARAMETER pcMethod        AS CHARACTER NO-UNDO.
  DEFINE INPUT  PARAMETER pcPath          AS CHARACTER NO-UNDO.
  DEFINE INPUT  PARAMETER plcRequestBody  AS LONGCHAR  NO-UNDO.
  DEFINE OUTPUT PARAMETER piStatusCode    AS INTEGER   NO-UNDO.
  DEFINE OUTPUT PARAMETER plcResponseBody AS LONGCHAR  NO-UNDO.
  DEFINE OUTPUT PARAMETER pcError         AS CHARACTER NO-UNDO.

  DEFINE VARIABLE oReqJson    AS JsonObject    NO-UNDO.
  DEFINE VARIABLE oHdrJson    AS JsonObject    NO-UNDO.
  DEFINE VARIABLE oRequest    AS IHttpRequest  NO-UNDO.
  DEFINE VARIABLE oResponse   AS IHttpResponse NO-UNDO.
  DEFINE VARIABLE oRespObj    AS JsonObject    NO-UNDO.
  DEFINE VARIABLE oRespArr    AS JsonArray     NO-UNDO.
  DEFINE VARIABLE oLib        AS IHttpClientLibrary NO-UNDO.
  DEFINE VARIABLE lcRaw       AS LONGCHAR      NO-UNDO.

  ASSIGN piStatusCode    = 0
         plcResponseBody = ""
         pcError         = "".

  RUN write-log(SUBSTITUTE("REQ method=&1 path=&2", CAPS(pcMethod), pcPath)).

  oReqJson = NEW JsonObject().
  oReqJson:ADD("proxyToken", gcProxyToken).
  oReqJson:ADD("method", CAPS(pcMethod)).
  oReqJson:ADD("path", pcPath).
  oReqJson:ADD("timeout", giDefaultTimeout).

  oHdrJson = NEW JsonObject().
  oHdrJson:ADD("Accept", "application/json").
  IF plcRequestBody <> "" THEN oHdrJson:ADD("Content-Type", "application/json").
  oReqJson:ADD("headers", oHdrJson).
  IF plcRequestBody <> "" THEN oReqJson:ADD("rawBody", plcRequestBody).

  oLib = ClientLibraryBuilder:Build():LIBRARY NO-ERROR.

  oRequest = RequestBuilder:POST(gcProxyUrl, oReqJson)
              :ContentType("application/json")
              :AcceptJson()
              :REQUEST.

  oResponse = ClientBuilder:Build():UsingLibrary(oLib):Client:Execute(oRequest) NO-ERROR.

  IF ERROR-STATUS:ERROR OR NOT VALID-OBJECT(oResponse) THEN DO:
    pcError = "Falha ao chamar proxy PHP.".
    RUN write-log("ERR " + pcError).
    RETURN.
  END.

  piStatusCode = oResponse:StatusCode.
  RUN write-log(SUBSTITUTE("RESP status=&1", piStatusCode)).

  IF NOT VALID-OBJECT(oResponse:Entity) THEN DO:
    RUN write-log("RESP body=(vazio)").
    RETURN.
  END.

  /* 1) Tenta JSON objeto */
  oRespObj = CAST(oResponse:Entity, JsonObject) NO-ERROR.
  IF VALID-OBJECT(oRespObj) THEN DO:
    oRespObj:WRITE(plcResponseBody, TRUE).
    RUN write-log("RESP body(json-obj) capturado").
    IF piStatusCode >= 400 THEN RUN save-response-dump(piStatusCode, plcResponseBody).
    RETURN.
  END.

  /* 2) Tenta JSON array */
  oRespArr = CAST(oResponse:Entity, JsonArray) NO-ERROR.
  IF VALID-OBJECT(oRespArr) THEN DO:
    oRespArr:WRITE(plcResponseBody, TRUE).
    RUN write-log("RESP body(json-arr) capturado").
    IF piStatusCode >= 400 THEN RUN save-response-dump(piStatusCode, plcResponseBody).
    RETURN.
  END.
   /*
  /* 3) Fallback: captura corpo cru (HTML, texto, etc.) */
  lcRaw = oResponse:Entity:GetText() NO-ERROR.
  plcResponseBody = lcRaw.
  RUN write-log("RESP body(raw) capturado").
  IF piStatusCode >= 400 THEN RUN save-response-dump(piStatusCode, plcResponseBody).
      */
END PROCEDURE.

FUNCTION decimalToJsonNumber RETURNS CHARACTER (INPUT pdValue AS DECIMAL):
  DEFINE VARIABLE cValue AS CHARACTER NO-UNDO.

  cValue = STRING(pdValue, "->>>>>>>>>9.99").
  cValue = TRIM(cValue).
  cValue = REPLACE(cValue, ".", "").
  cValue = REPLACE(cValue, ",", ".").

  RETURN cValue.
END FUNCTION.

FUNCTION jsonEscape RETURNS CHARACTER (INPUT pc AS CHARACTER):
  DEFINE VARIABLE c AS CHARACTER NO-UNDO.
  c = REPLACE(pc, "~\", "\\").
  c = REPLACE(c, "~"", "\~"").
  c = REPLACE(c, CHR(10), "\n").
  c = REPLACE(c, CHR(13), "\r").
  c = REPLACE(c, CHR(9),  "\t").
  RETURN c.
END FUNCTION.

/* Busca ID da Tabela de Preço pelo nrtabpre */
PROCEDURE findPriceTableIdByNrtabpre:
  DEFINE INPUT  PARAMETER pcNrtabpre AS CHARACTER NO-UNDO.
  DEFINE OUTPUT PARAMETER piId       AS INTEGER   NO-UNDO.

  DEFINE VARIABLE iStatus AS INTEGER   NO-UNDO.
  DEFINE VARIABLE lcResp  AS LONGCHAR  NO-UNDO.
  DEFINE VARIABLE cErr    AS CHARACTER NO-UNDO.

  DEFINE VARIABLE oParser AS Progress.Json.ObjectModel.ObjectModelParser NO-UNDO.
  DEFINE VARIABLE oArr    AS Progress.Json.ObjectModel.JsonArray         NO-UNDO.
  DEFINE VARIABLE oObj    AS Progress.Json.ObjectModel.JsonObject        NO-UNDO.

  DEFINE VARIABLE i     AS INTEGER   NO-UNDO.
  DEFINE VARIABLE cCode AS CHARACTER NO-UNDO.

  piId = 0.
  IF TRIM(pcNrtabpre) = "" THEN RETURN.

  RUN portalCall(
    "GET",
    "/api/base/price-tables?q=" + urlEncode(pcNrtabpre),
    "",
    OUTPUT iStatus,
    OUTPUT lcResp,
    OUTPUT cErr
  ).

  IF cErr <> "" OR iStatus <> 200 OR lcResp = "" OR lcResp = ? THEN RETURN.

  oParser = NEW Progress.Json.ObjectModel.ObjectModelParser().
  oArr = CAST(oParser:Parse(lcResp), Progress.Json.ObjectModel.JsonArray) NO-ERROR.
  IF ERROR-STATUS:ERROR OR NOT VALID-OBJECT(oArr) THEN RETURN.

  DO i = 1 TO oArr:Length:
    oObj = oArr:GetJsonObject(i) NO-ERROR.
    IF ERROR-STATUS:ERROR OR NOT VALID-OBJECT(oObj) THEN NEXT.

    cCode = oObj:GetCharacter("nrtabpre") NO-ERROR.
    IF ERROR-STATUS:ERROR THEN NEXT.

    IF TRIM(cCode) = TRIM(pcNrtabpre) THEN DO:
      piId = oObj:GetInteger("id") NO-ERROR.
      IF ERROR-STATUS:ERROR THEN piId = 0.
      RETURN.
    END.
  END.
END PROCEDURE.

/* Upsert de Tabela de Preço */
PROCEDURE upsertPriceTable:
  DEFINE INPUT  PARAMETER pcNrtabpre  AS CHARACTER NO-UNDO.
  DEFINE INPUT  PARAMETER pcDescricao AS CHARACTER NO-UNDO.
  DEFINE INPUT  PARAMETER piSituacao  AS INTEGER   NO-UNDO.
  DEFINE OUTPUT PARAMETER piId        AS INTEGER   NO-UNDO.

  DEFINE VARIABLE lcBody  AS LONGCHAR  NO-UNDO.
  DEFINE VARIABLE lcResp  AS LONGCHAR  NO-UNDO.
  DEFINE VARIABLE cErr    AS CHARACTER NO-UNDO.
  DEFINE VARIABLE iStatus AS INTEGER   NO-UNDO.

  DEFINE VARIABLE oParser AS Progress.Json.ObjectModel.ObjectModelParser NO-UNDO.
  DEFINE VARIABLE oObj    AS Progress.Json.ObjectModel.JsonObject        NO-UNDO.
  DEFINE VARIABLE iRespId AS INTEGER NO-UNDO.

  piId = 0.
  IF TRIM(pcNrtabpre) = "" THEN RETURN.

  RUN findPriceTableIdByNrtabpre(pcNrtabpre, OUTPUT piId).

  IF piId = 0 THEN DO:
    lcBody = SUBSTITUTE(
      '~{"nrtabpre":"&1","descricao":"&2","situacao":&3~}',
      jsonEscape(pcNrtabpre),
      jsonEscape(pcDescricao),
      piSituacao
    ).

    RUN portalCall("POST", "/api/base/price-tables", lcBody, OUTPUT iStatus, OUTPUT lcResp, OUTPUT cErr).
    IF cErr <> "" THEN UNDO, THROW NEW Progress.Lang.AppError(cErr, 0).

    IF iStatus = 201 THEN DO:
      iRespId = 0.
      oParser = NEW Progress.Json.ObjectModel.ObjectModelParser().
      oObj = CAST(oParser:Parse(lcResp), Progress.Json.ObjectModel.JsonObject) NO-ERROR.
      IF NOT ERROR-STATUS:ERROR AND VALID-OBJECT(oObj) THEN DO:
        iRespId = oObj:GetInteger("id") NO-ERROR.
        IF ERROR-STATUS:ERROR THEN iRespId = 0.
      END.

      IF iRespId > 0 THEN piId = iRespId.
      ELSE RUN findPriceTableIdByNrtabpre(pcNrtabpre, OUTPUT piId).

      IF piId = 0 THEN UNDO, THROW NEW Progress.Lang.AppError("Não localizou a tabela criada: " + pcNrtabpre, 0).
      RETURN.
    END.

    IF iStatus = 409 THEN DO:
      RUN findPriceTableIdByNrtabpre(pcNrtabpre, OUTPUT piId).
      IF piId = 0 THEN UNDO, THROW NEW Progress.Lang.AppError("Tabela já existe, mas não localizou por nrtabpre: " + pcNrtabpre, 0).
    END.
    ELSE DO:
      UNDO, THROW NEW Progress.Lang.AppError("Erro POST price-tables: " + STRING(iStatus) + " " + STRING(lcResp), 0).
    END.
  END.

  lcBody = SUBSTITUTE('~{"descricao":"&1","situacao":&2~}', jsonEscape(pcDescricao), piSituacao).
  RUN portalCall("PATCH", "/api/base/price-tables/" + STRING(piId), lcBody, OUTPUT iStatus, OUTPUT lcResp, OUTPUT cErr).
  IF cErr <> "" THEN UNDO, THROW NEW Progress.Lang.AppError(cErr, 0).
  IF iStatus <> 200 THEN UNDO, THROW NEW Progress.Lang.AppError("Erro PATCH price-tables/[id]: " + STRING(iStatus) + " " + STRING(lcResp), 0).

  iRespId = 0.
  oParser = NEW Progress.Json.ObjectModel.ObjectModelParser().
  oObj = CAST(oParser:Parse(lcResp), Progress.Json.ObjectModel.JsonObject) NO-ERROR.
  IF NOT ERROR-STATUS:ERROR AND VALID-OBJECT(oObj) THEN DO:
    iRespId = oObj:GetInteger("id") NO-ERROR.
    IF ERROR-STATUS:ERROR THEN iRespId = 0.
  END.
  IF iRespId > 0 THEN piId = iRespId.
END PROCEDURE.

/* Upsert de vários Itens por SKU na Tabela de Preço */
PROCEDURE upsertPriceTableItemBySku:
  DEFINE INPUT PARAMETER piPriceTableId AS INTEGER  NO-UNDO.
  DEFINE INPUT PARAMETER poItems        AS JsonArray NO-UNDO.

  DEFINE VARIABLE lcBody AS LONGCHAR NO-UNDO.

/* Busca ID do Tipo de Pedido pelo código */
PROCEDURE findOrderTypeIdByCod:
  DEFINE INPUT  PARAMETER piCod AS INTEGER NO-UNDO.
  DEFINE OUTPUT PARAMETER piId  AS INTEGER NO-UNDO.

  piId = 0.
  EMPTY TEMP-TABLE ttOrderType.

  RUN portalCall("GET", "/api/base/order-types?q=" + STRING(piCod), "", OUTPUT iStatus, OUTPUT lcResp, OUTPUT cErr).
  IF cErr <> "" THEN UNDO, THROW NEW Progress.Lang.AppError(cErr, 0).
  IF iStatus <> 200 THEN UNDO, THROW NEW Progress.Lang.AppError("Erro GET order-types: " + STRING(iStatus) + " " + STRING(lcResp), 0).

  TEMP-TABLE ttOrderType:READ-JSON("LONGCHAR", lcResp, "EMPTY").

  FIND FIRST ttOrderType WHERE ttOrderType.codtipoped = piCod NO-ERROR.
  IF AVAILABLE ttOrderType THEN piId = ttOrderType.id.
END PROCEDURE.

/* Upsert de Tipo de Pedido */
PROCEDURE upsertOrderType:
  DEFINE INPUT  PARAMETER piCod       AS INTEGER   NO-UNDO.
  DEFINE INPUT  PARAMETER pcDescricao AS CHARACTER NO-UNDO.
  DEFINE INPUT  PARAMETER piSituacao  AS INTEGER   NO-UNDO.
  DEFINE OUTPUT PARAMETER piId        AS INTEGER   NO-UNDO.

  DEFINE VARIABLE lcBody AS LONGCHAR NO-UNDO.

  RUN findOrderTypeIdByCod(piCod, OUTPUT piId).

  IF piId = 0 THEN DO:
    lcBody = SUBSTITUTE(
      '~{"codtipoped":&1,"descricao":"&2","situacao":&3~}',
      piCod,
      jsonEscape(pcDescricao),
      piSituacao
    ).
    RUN portalCall("POST", "/api/base/order-types", lcBody, OUTPUT iStatus, OUTPUT lcResp, OUTPUT cErr).
    IF cErr <> "" THEN UNDO, THROW NEW Progress.Lang.AppError(cErr, 0).
    IF iStatus <> 201 THEN UNDO, THROW NEW Progress.Lang.AppError("Erro POST order-types: " + STRING(iStatus) + " " + STRING(lcResp), 0).

    RUN findOrderTypeIdByCod(piCod, OUTPUT piId).
    IF piId = 0 THEN UNDO, THROW NEW Progress.Lang.AppError("Não localizou o tipo de pedido criado: " + STRING(piCod), 0).
  END.
  ELSE DO:
    lcBody = SUBSTITUTE('~{"descricao":"&1","situacao":&2~}', jsonEscape(pcDescricao), piSituacao).
    RUN portalCall("PATCH", "/api/base/order-types/" + STRING(piId), lcBody, OUTPUT iStatus, OUTPUT lcResp, OUTPUT cErr).
    IF cErr <> "" THEN UNDO, THROW NEW Progress.Lang.AppError(cErr, 0).
    IF iStatus <> 200 THEN UNDO, THROW NEW Progress.Lang.AppError("Erro PATCH order-types/[id]: " + STRING(iStatus) + " " + STRING(lcResp), 0).
  END.
END PROCEDURE.
